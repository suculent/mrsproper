export type TargetState =
  | { step: 'idle' }
  | { step: 'resolving' }
  | { step: 'resolved'; target: ResolvedTargetDTO }
  | { step: 'confirmed'; target: ResolvedTargetDTO }
  | { step: 'error'; code: TargetErrorCode; input: string }
