    _classifyDialog(dialog): inspects dialog.entity using instanceof Api.Chat / Api.Channel; stores entity in _dialogMap keyed by entity.id.toString(); returns AdminDialogDTO with id=entity.id.toString(), title=(dialog as any).title, avatar=null (deferred — avatar fetch would block enumeration), memberCount extracted from entity fields, type, canAct, greyReason.     getState(): returns _currentState.
    getDialogEntity(dialogId: string): returns _dialogMap.get(dialogId) ?? null.
    refresh(): clears _dialogMap, re-calls enumerateAtLogin().
