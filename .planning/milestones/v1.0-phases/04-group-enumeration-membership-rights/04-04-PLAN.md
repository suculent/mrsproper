    MembershipScannerClient interface: { invoke(r: unknown): Promise<unknown> }
    MembershipScannerOptions interface: { client: MembershipScannerClient; governor: RateLimitGovernor; enumerator: GroupEnumerator; resolver: { getResolvedUser(): { id: bigint; accessHash: bigint } | null }; onStateChange: (s: ScanState) => void }

    _checkMembership(dialog: AdminDialogDTO, inputUser: Api.InputPeerUser): Promise<boolean>: gets entity from _enumerator.getDialogEntity(dialog.id); for Api.Channel entities: invoke GetParticipant wrapped in governor.run; inspect participant subtype for Left/Banned; catch USER_NOT_PARTICIPANT → return false; for Api.Chat entities: invoke GetFullChat wrapped in governor.run; scan participants array by .userId.toString() === resolved.id.toString(); return false for forbidden/unfound.

    abortScan(): increments _generation, calls abort, emits idle.
    getState(): returns _currentState.
    retryGroup(dialogId: string): re-runs _checkMembership for the single group; merges result into current complete state's results array; emits updated complete state.
