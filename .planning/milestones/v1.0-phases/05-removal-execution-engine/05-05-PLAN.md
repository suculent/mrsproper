export function registerSearchChannels(win, searcher, resolver): void {
  ipcMain.handle(SEARCH_START, async (_event, payload: unknown) => {
    const parsed = SearchStartSchema.parse(payload)  // throws ZodError on invalid
    searcher.search(parsed.query).catch(() => {})
    return searcher.getState()
  })
  // Two-layer security gate: Zod first, then in-memory candidate map
  ipcMain.handle(TARGET_RESOLVE_CANDIDATE, async (_event, payload: unknown) => {
    const parsed = ResolveCandidateSchema.parse(payload)
    const accessHash = searcher.getCandidateAccessHash(parsed.userId)
    if (accessHash === null) {
      throw new Error('Unknown userId — not in current candidate set')
    }
    // ...
  })
}
export function emitSearchState(win: BrowserWindow, state: SearchState): void {
  if (!win.isDestroyed()) {
    win.webContents.send(SEARCH_STATE, state)
  }
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| renderer → IPC → removal:start | Renderer cannot inject targetId or accessHash — executor reads those from resolver.getResolvedUser() only |
| renderer → IPC → dialogIds payload | Renderer-supplied dialogIds validated against enumerator.getState().dialogs before dispatch |
| IPC → executor.execute() | Confirmed flag is enforced by the confirm dialog in the renderer (UI gate); engine layer also requires auth gate at IPC boundary |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-05-SC | Tampering | npm/pip/cargo installs | accept | No new npm packages |
| T-05-05-A | Tampering | Renderer injects fake dialogIds | mitigate | Zod bounds check (max 500) + enumerator.getState().dialogs membership validation in removal:start handler before executor dispatch |
| T-05-05-B | Elevation of Privilege | Renderer bypasses confirm dialog via IPC | mitigate | Engine starts on first removal:start invoke; the confirm dialog in the renderer is the UI gate; no additional IPC "confirmed" flag needed because the engine itself is harmless without a valid auth state + valid dialogIds in the enumerated set |
| T-05-05-C | Tampering | severity field injection ('nuke' or other) | mitigate | Zod z.enum(['kick','ban']) at IPC boundary |
| T-05-05-D | Denial of Service | TARGET_RESET while removal running | mitigate | CR-01 ordering: executor.abortExecution() fires first in TARGET_RESET handler; executor checkpoints and stops cleanly |
</threat_model>
